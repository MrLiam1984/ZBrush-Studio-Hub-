    function updateDojoTimerDisplay() {
      const m = String(Math.floor(dojoTimerRemainingSec / 60)).padStart(2, '0');
      const s = String(dojoTimerRemainingSec % 60).padStart(2, '0');
      document.getElementById('dojoTimerDisplay').textContent = `${m}:${s}`;
    }

    function toggleDojoTimer() {
      if (dojoTimerRunning) {
        pauseDojoTimer();
      } else {
        dojoTimerRunning = true;
        document.getElementById('dojoTimerStartBtn').innerHTML = `Tạm dừng`;
        dojoTimerInterval = setInterval(() => {
          if (dojoTimerRemainingSec > 0) {
            dojoTimerRemainingSec--;
            updateDojoTimerDisplay();
          } else {
            completeDojoTimerEarly(true);
          }
        }, 1000);
      }
    }

    function pauseDojoTimer() {
      dojoTimerRunning = false;
      if (dojoTimerInterval) clearInterval(dojoTimerInterval);
      const btn = document.getElementById('dojoTimerStartBtn');
      if (btn) btn.innerHTML = `Bắt đầu`;
    }

    function resetDojoTimer() {
      pauseDojoTimer();
      dojoTimerRemainingSec = dojoTimerInitialMin * 60;
      updateDojoTimerDisplay();
    }

    function completeDojoTimerEarly(fullCompleted = false) {
      pauseDojoTimer();
      const elapsedMinutes = fullCompleted
        ? dojoTimerInitialMin
        : Math.max(5, Math.round((dojoTimerInitialMin * 60 - dojoTimerRemainingSec) / 60));
      studioState.totalMinutesPracticed = (studioState.totalMinutesPracticed || 0) + elapsedMinutes;
      dojoTimerRemainingSec = dojoTimerInitialMin * 60;
      updateDojoTimerDisplay();
      addXP(elapsedMinutes, `Hoàn thành ${elapsedMinutes} phút luyện Speed Sculpt tập trung!`);
    }

    function initThreeViewportIfNeeded() {
      if (threeRenderer) return;
      const container = document.getElementById('threeCanvasContainer');
      if (!container || typeof THREE === 'undefined') return;

      const w = container.clientWidth || 720;
      const h = container.clientHeight || 480;

      threeScene = new THREE.Scene();
      threeScene.background = new THREE.Color(0x121622);

      // Default 85mm equivalent perspective FOV (~24 deg vertical FOV)
      threeCamera = new THREE.PerspectiveCamera(24, w / h, 0.1, 1000);
      threeCamera.position.set(0, 1.2, 6.2);

      threeRenderer = new THREE.WebGLRenderer({ antialias: true, preserveDrawingBuffer: true });
      threeRenderer.setSize(w, h);
      threeRenderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      container.appendChild(threeRenderer.domElement);

      if (THREE.OrbitControls) {
        threeControls = new THREE.OrbitControls(threeCamera, threeRenderer.domElement);
        threeControls.enableDamping = true;
        threeControls.dampingFactor = 0.06;
        threeControls.target.set(0, 0.4, 0);
      }

      // Studio 3-Point Lighting
      const ambient = new THREE.AmbientLight(0xffffff, 0.45);
      threeScene.add(ambient);

      threeKeyLight = new THREE.DirectionalLight(0xfff4ea, 1.15);
      threeKeyLight.position.set(4, 6, 4);
      threeScene.add(threeKeyLight);

      const fillLight = new THREE.DirectionalLight(0x88b4ff, 0.45);
      fillLight.position.set(-5, 2, 2);
      threeScene.add(fillLight);

      threeRimLight = new THREE.DirectionalLight(0x38bdf8, 0.85);
      threeRimLight.position.set(0, 4, -6);
      threeScene.add(threeRimLight);

      threeGridHelper = new THREE.GridHelper(6, 20, 0xff6b2b, 0x2a3246);
      threeGridHelper.position.y = -1.2;
      threeScene.add(threeGridHelper);

      setupViewportDragAndDrop();
      loadProceduralDemoModel('anime_head');

      function animateThree() {
        requestAnimationFrame(animateThree);
        if (threeIsTurntable && threeCurrentModel) {
          threeCurrentModel.rotation.y += 0.008;
        }
        if (threeControls) threeControls.update();
        if (threeRenderer && threeScene && threeCamera) {
          threeRenderer.render(threeScene, threeCamera);
        }
      }
      animateThree();

      window.addEventListener('resize', resizeThreeViewport);
    }

    function resizeThreeViewport() {
      const container = document.getElementById('threeCanvasContainer');
      if (!container || !threeRenderer || !threeCamera) return;
      const w = container.clientWidth;
      const h = container.clientHeight;
      if (w === 0 || h === 0) return;
      threeCamera.aspect = w / h;
      threeCamera.updateProjectionMatrix();
      threeRenderer.setSize(w, h);
    }

    function createShadingMaterial(mode) {
      if (mode === 'silhouette') {
        return new THREE.MeshBasicMaterial({
          color: 0x050608,
          wireframe: threeIsWireframe
        });
      }
      if (mode === 'greyclay') {
        return new THREE.MeshStandardMaterial({
          color: 0x94a3b8,
          roughness: 0.55,
          metalness: 0.08,
          wireframe: threeIsWireframe,
          flatShading: false
        });
      }
      if (mode === 'animeskin') {
        return new THREE.MeshPhysicalMaterial({
          color: 0xffd8c4,
          emissive: 0x2b0e08,
          roughness: 0.38,
          metalness: 0.02,
          clearcoat: 0.15,
          wireframe: threeIsWireframe
        });
      }
      // Default: ZBrush Red Wax Clay
      return new THREE.MeshPhysicalMaterial({
        color: 0xc84b31,
        emissive: 0x240703,
        roughness: 0.35,
        metalness: 0.05,
        clearcoat: 0.25,
        clearcoatRoughness: 0.3,
        wireframe: threeIsWireframe
      });
    }

    function applyMaterialToCurrentModel() {
      if (!threeCurrentModel) return;
      const mat = createShadingMaterial(threeCurrentShading);
      threeCurrentModel.traverse(child => {
        if (child.isMesh) {
          if (child.userData.isFlatPlanar && threeCurrentShading !== 'silhouette') {
            mat.flatShading = true;
          }
          child.material = mat;
        }
      });
    }

    function setViewportShading(mode) {

        }
      });
      applyMaterialToCurrentModel();
    }

    function process3DFileObject(file) {
      initThreeViewportIfNeeded();
      const ext = file.name.split('.').pop().toLowerCase();
      const reader = new FileReader();

      showToast(`Đang đọc file 3D "${file.name}"...`, 'info');

      if (ext === 'obj') {
        reader.onload = e => {
          try {
            const loader = new THREE.OBJLoader();
            const obj = loader.parse(e.target.result);
            mountCustom3DObject(obj, file.name);
          } catch (err) {
            showToast('Lỗi đọc file .OBJ, hãy kiểm tra định dạng xuất từ ZBrush/Maya!', 'info');
          }
        };
        reader.readAsText(file);
      } else if (ext === 'glb' || ext === 'gltf') {
        reader.onload = e => {
          const loader = new THREE.GLTFLoader();
          loader.parse(e.target.result, '', gltf => {
            mountCustom3DObject(gltf.scene, file.name);
          }, () => {
            showToast('Không thể đọc file GLB/GLTF này!', 'info');
          });
        };
        reader.readAsArrayBuffer(file);
      } else if (ext === 'stl') {
        reader.onload = e => {
          const loader = new THREE.STLLoader();
          const geom = loader.parse(e.target.result);
          const mesh = new THREE.Mesh(geom);
          const grp = new THREE.Group();
          grp.add(mesh);
          mountCustom3DObject(grp, file.name);
        };
        reader.readAsArrayBuffer(file);
      } else {
        showToast('Vui lòng chọn file .OBJ, .GLB, .GLTF hoặc .STL xuất từ ZBrush/Maya!', 'info');
      }
    }

    function mountCustom3DObject(object3D, fileName) {
      if (threeCurrentModel) threeScene.remove(threeCurrentModel);

      // Auto-center and normalize scale to fit viewport nicely
      const box = new THREE.Box3().setFromObject(object3D);
      const size = new THREE.Vector3();
      const center = new THREE.Vector3();
      box.getSize(size);
      box.getCenter(center);

      const maxDim = Math.max(size.x, size.y, size.z, 0.001);
      const scaleFactor = 2.4 / maxDim;
      object3D.scale.setScalar(scaleFactor);

      // Recompute center after scale
      const scaledBox = new THREE.Box3().setFromObject(object3D);
      scaledBox.getCenter(center);
      object3D.position.sub(center);
      object3D.position.y += 0.3;

      threeCurrentModel = object3D;
      threeScene.add(threeCurrentModel);
      applyMaterialToCurrentModel();
      resetViewportCamera();

      document.getElementById('hudModelName').textContent = `File: ${fileName}`;
      runMeshAuditOnObject(threeCurrentModel, size);
      addXP(30, `Nạp & Kiểm tra file 3D: ${fileName}`);
    }

    function runMeshAuditOnObject(object3D, originalSize = null) {
      let totalVerts = 0;
      let totalFaces = 0;
      let hasUV = false;

      object3D.traverse(child => {
        if (child.isMesh && child.geometry) {
          const geom = child.geometry;
          const pos = geom.attributes.position;
          if (pos) {
            totalVerts += pos.count;
            if (geom.index) {
              totalFaces += Math.round(geom.index.count / 3);
            } else {
              totalFaces += Math.round(pos.count / 3);
            }
          }
          if (geom.attributes.uv) hasUV = true;
        }
      });

      const box = new THREE.Box3().setFromObject(object3D);
      const sz = originalSize || new THREE.Vector3();
      if (!originalSize) box.getSize(sz);

      document.getElementById('auditVertices').textContent = totalVerts.toLocaleString();
      document.getElementById('auditFaces').textContent = totalFaces.toLocaleString();
      document.getElementById('auditBBox').textContent = `${sz.x.toFixed(2)} x ${sz.y.toFixed(2)} x ${sz.z.toFixed(2)}`;
      const uvEl = document.getElementById('auditUVStatus');
      uvEl.textContent = hasUV ? 'Đã có UV Map' : 'Chưa có UV (Dynamesh)';
      uvEl.className = hasUV ? 'text-xs font-bold text-emerald-400 mt-1' : 'text-xs font-bold text-amber-400 mt-1';

      const recBox = document.getElementById('auditRecommendationBox');
      let densityTag = '';
      if (totalFaces < 15000) {
        densityTag = `Mật độ Low/Mid-Poly (${totalFaces.toLocaleString()} Tris): Lưới rất nhẹ và sạch, phù hợp để kiểm tra Blockout hoặc đưa sang Maya UV / Substance Painter.`;
      } else if (totalFaces < 150000) {
        densityTag = `Mật độ Sculpt Trung Bình (${totalFaces.toLocaleString()} Tris): Phù hợp giai đoạn Secondary Forms (nặn ngũ quan, cơ thể, tóc CurveTubes).`;
      } else {
        densityTag = `High-Poly Sculpt (${totalFaces.toLocaleString()} Tris): Lưới dày chi tiết cao. Nhớ chạy ZRemesher hoặc Quad Draw trước khi đưa vào Game Engine.`;
      }
      recBox.innerHTML = `
${densityTag}
💡 Mẹo: Bấm nút Silhouette trên thanh công cụ để kiểm tra xem dáng nhân vật có bị dính khối không.
`;
    }

    function submitLightboxComment() {
      if (!activeLightboxItemId) return;
      const item = (studioState.portfolioItems || []).find(i => i.id === activeLightboxItemId);
      if (!item) return;

      const author = document.getElementById('lightboxCommentAuthor').value.trim() || 'Mentor / Bạn học';
      const text = document.getElementById('lightboxCommentText').value.trim();
      if (!text) return;

      if (!item.comments) item.comments = [];
      item.comments.push({ author, text });
      document.getElementById('lightboxCommentText').value = '';
      saveStudioState();
      renderLightboxComments(item);
      showToast('Đã thêm nhận xét vào tác phẩm!', 'success');
    }

    function exportStudioJSON() {
      const dataStr = JSON.stringify(studioState, null, 2);
      const blob = new Blob([dataStr], { type: 'application/json' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `zbrush_studio_backup_${new Date().toISOString().slice(0, 10)}.json`;
      document.body.appendChild(a);
      a.click();
      a.remove();
      URL.revokeObjectURL(url);
      showToast('Đã tải xuống file Backup (.JSON) thành công!', 'success');
    }

    function importStudioJSON(event) {
      const file = event.target.files?.[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = e => {
        try {
          const parsed = JSON.parse(e.target.result);
          if (parsed && parsed.chapters) {
            studioState = { ...studioState, ...parsed };
            saveStudioState();
            updateAllUI();
            closeModal('backupModal');
            showToast('Đã khôi phục toàn bộ dữ liệu học tập & Portfolio!', 'success');
          }
        } catch (err) {
          showToast('File JSON không hợp lệ!', 'info');
        }
      };
      reader.readAsText(file);
    }
